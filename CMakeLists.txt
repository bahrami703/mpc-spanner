cmake_minimum_required(VERSION 3.16)
project(lqt VERSION 1.0.0 LANGUAGES CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

if(NOT DEFINED CMAKE_CXX_EXTENSIONS)
  set(CMAKE_CXX_EXTENSIONS OFF)
endif()

option(LQT_NATIVE "Optimize the tools and tests for the build machine" OFF)
option(LQT_BUILD_TOOLS "Build the lqt-spanner command-line tool" ON)
option(LQT_BUILD_TESTS "Build the tests" ON)

add_library(lqt INTERFACE)
add_library(lqt::lqt ALIAS lqt)
target_include_directories(lqt INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_features(lqt INTERFACE cxx_std_20)

set(LQT_PROGRAM_OPTIONS "")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
  list(APPEND LQT_PROGRAM_OPTIONS -Wall -Wextra)
  if(LQT_NATIVE)
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag(-march=native LQT_HAS_MARCH_NATIVE)
    check_cxx_compiler_flag(-mcpu=native LQT_HAS_MCPU_NATIVE)
    if(LQT_HAS_MARCH_NATIVE)
      list(APPEND LQT_PROGRAM_OPTIONS -march=native)
    elseif(LQT_HAS_MCPU_NATIVE)
      list(APPEND LQT_PROGRAM_OPTIONS -mcpu=native)
    endif()
  endif()
endif()

if(LQT_BUILD_TOOLS)
  add_executable(lqt-spanner tools/lqt_spanner.cpp)
  target_link_libraries(lqt-spanner PRIVATE lqt::lqt)
  target_compile_options(lqt-spanner PRIVATE ${LQT_PROGRAM_OPTIONS})
endif()

if(LQT_BUILD_TESTS)
  enable_testing()
  add_executable(lqt-tests tests/lqt_tests.cpp)
  target_link_libraries(lqt-tests PRIVATE lqt::lqt)
  target_compile_options(lqt-tests PRIVATE ${LQT_PROGRAM_OPTIONS})
  add_test(NAME lqt-tests COMMAND lqt-tests)
endif()

include(GNUInstallDirs)
install(DIRECTORY include/lqt DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(TARGETS lqt EXPORT lqtTargets)
install(EXPORT lqtTargets NAMESPACE lqt:: FILE lqtConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/lqt)
if(LQT_BUILD_TOOLS)
  install(TARGETS lqt-spanner RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
